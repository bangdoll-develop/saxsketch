"""B-4a プローブ 1: 径 2 本の**通る範囲そのもの**を直接測る。

1 段外の理由しか出ない。ここは帯を無視して総当たりし、`_cheap_ok`（段 1〜2）が
通る区間の端を出す。相手（shank_dia / bore_dia / body_dia）を振って、壁が
数字ではなく関係であることも同時に見る。
"""
import sys
from pathlib import Path

from saxsketch.core.registry import get_registry
import sweep_limits as sl

SIZES = ("soprano", "alto", "tenor")


def span(comp, base, key, lo, hi, step):
    """[lo, hi] を step 刻みで舐め、段 1〜2 が通る値の最小/最大と理由。"""
    ok = []
    why_lo = why_hi = ""
    v = lo
    while v <= hi + 1e-9:
        bad = sl._cheap_ok(comp, dict(base, **{key: round(v, 3)}))
        if bad is None:
            ok.append(round(v, 3))
        else:
            if not ok:
                why_lo = bad
            else:
                why_hi = why_hi or bad
        v += step
    if not ok:
        return None, None, why_lo, why_hi
    return ok[0], ok[-1], why_lo, why_hi

def model_ok(comp, p):
    err = comp.params_error(p)
    if err is not None:
        return "params_error: " + err[:90]
    try:
        pf.cavity_stations(p)
    except Exception as exc:                                  # noqa: BLE001
        return "%s: %s" % (type(exc).__name__, str(exc)[:80])
    return None


def model_span(comp, base, key, lo, hi, step):

        bad = model_ok(comp, dict(base, **{key: round(v, 4)}))
        if bad is None:
            ok.append(round(v, 4))
        elif not ok:
            wl = bad
        else:
            wh = wh or bad
        v += step
    return (ok[0] if ok else None, ok[-1] if ok else None, wl, wh)

        "overall_length_hard_min": round(bl + 6.5, 1),
        "overall_length_hard_max": round(1.8 * L, 1),
        "body_dia_hard_max": 54.0,
        "shank_dia_hard_max": 54.0,
    }
    print("  掛け金の候補（単独で置いたとき段 1〜2 が通るか）:", flush=True)
    for name, v in cands.items():
        key = name.rsplit("_hard_", 1)[0]
        p = dict(base, **{key: v})
        # 径 2 本は相手も一緒に動かさないと関係で弾かれる。相手を壁側へ寄せる。
        if key == "body_dia" and v < base["shank_dia"]:
            p["shank_dia"] = round(base["bore_dia"] - 11.0 + 0.1, 2)
        if key == "shank_dia" and v > base["body_dia"]:
            p["body_dia"] = 54.0
        if key == "shank_dia" and v < base["bore_dia"]:
            p["bore_dia"] = round(base["bore_dia"] - 11.0, 2)
        print("    %-26s %8s -> %s" % (name, v, model_ok(comp, p) or "ok"),
              flush=True)
    print(flush=True)

# -*- coding: utf-8 -*-
"""入れた壁が (1) 通る (2) 1 段外で断られる を確かめる。"""
import sys
from pathlib import Path
sys.path.insert(0, str(Path.home() / "mp_design"))
from saxsketch.core.registry import get_registry
from saxsketch.instruments.saxophone.components import profiles as pf

KEYS = ("body_dia", "shank_dia", "shank_length", "shoulder_ratio",
        "overall_length")
for size in ("soprano", "alto", "tenor"):
    comp = get_registry().get("saxophone").make_component("mouthpiece", size)
    base = {s.key: s.default for s in comp.param_specs()}
    sp = {s.key: s for s in comp.param_specs()}
    print("== %s ==" % size, flush=True)
    print("  既定 params_error: %s" % comp.params_error(base), flush=True)
    print("  既定 warnings: %s" % comp.warnings(base), flush=True)
    for k in KEYS:
        s = sp[k]
        lo, hi = s.bounds()
        print("  %-15s 帯 %g〜%g / 壁 %s〜%s (hard %s/%s)"
              % (k, s.min, s.max, lo, hi, s.hard_min, s.hard_max), flush=True)
        for tag, v, partner in (("壁下", lo, True), ("壁上", hi, False)):
            p = dict(base, **{k: v})
            # 径 2 本の掛け金は相手を壁側へ寄せないと関係で弾かれる（設計どおり）
            if partner and k in ("body_dia", "shank_dia"):
                p["bore_dia"] = sp["bore_dia"].hard_min
                p["shank_dia"] = sp["shank_dia"].hard_min
                if k == "body_dia":
                    p["body_dia"] = v
            if not partner and k == "shank_dia":
                p["body_dia"] = sp["body_dia"].hard_max
            err = comp.params_error(p)
            print("    %s %-8s -> %s" % (tag, v, (err or "ok")[:80]), flush=True)
    # 床の 1 段外
    for k, step in (("shank_length", 0.5), ("shoulder_ratio", 0.005)):
        cross = (pf.shank_length_at_heel_floor(base) if k == "shank_length"
                 else pf.shoulder_ratio_at_heel_floor(base))
        for d in (0.0, +step):
            v = round(cross + d, 4)
            err = comp.params_error(dict(base, **{k: v}))
            print("    床 %s=%.4f (cross %.4f) -> %s"
                  % (k, v, cross, (err or "ok")[:60]), flush=True)
    print(flush=True)
