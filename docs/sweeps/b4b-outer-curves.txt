# Phase B-4b 外形・曲がりとビーク 3 本の生ログ（2026-09-07、3 サイズ・OCCT ビルドあり）
# 読み方は docs/param-limits.md の「B-4b 外形・曲がりとビーク」。
# 掃引器: PYTHONPATH=. python cad/windycity/sweep_limits.py <key> [--steps N]
#
# **このスライスは掃引器では壁が測れなかった。** 3 本とも段 1〜3（成立条件 /
# ステーション / OCCT ソリッド 1 個）が帯のはるか外まで ok を返す:
#
#   body_waist        ±0.36 まで ok（頂点が壊れるのは +0.0726 ＝ 5 倍内側）
#   shoulder_fullness ±4.0  まで ok（えぐれ始めるのは −0.373 ＝ 10 倍内側）
#   beak_height       +48.8〜53.1 まで ok（定義が壊れるのは +18.1〜23.0）
#
# 壊れるのは形ではなく**主張**（母線は単調 ／ beak_height は口の開き量）なので、
# 形を直に測る側を作って初めて見えた。節の並びはその順:
#
#   §1  壁を入れる前の掃引 4 本（body_waist ／ body_waist --axis outer_r ／
#       shoulder_fullness ／ beak_height）。**どれも壁を見つけていない**のが結果。
#   §2  掃引器の beak_y 軸を直した後の beak_height（100% → 41〜48%）。
#   §3  過去の掃引と比較できることの確認（tip_opening を回し直して
#       committed の b0-remeasure.txt と突き合わせ）。
#   §4  プローブ 1: 頂点／えぐれと OCCT のソリッド数を壁の内外で。
#   §5  プローブ 2: 単調性の目 2 つを足して測り直し、beak_height が割れる点を挟む。
#   §6  壁を入れた後の掃引 3 本（今度は掃引器も止まる）。
#   §7  sweep_outer.py（主張 (1) を outer_apex へ差し替えた後）。
#
# 途中で道具の穴が 3 つ出た。B-4a の「測った理由を捨てる」型とは別で、
# どれも **道具が式を測っていて形を測っていない**:
#   (1) sweep_outer.py の主張 (1) が pf.heel_z を比べていた。heel_z は
#       (全長 − シャンク長)/(1 + 肩比) の軸方向の式なので、body_waist /
#       shoulder_fullness では定義上動かない ——いちばん頂点を壊す 2 本に
#       対して構造的に無力だった。outer_apex（形の頂点）へ差し替え。
#   (2) その主張の輪は **params_error に弾かれた組も回していた**（_cases は
#       わざと成立しない組も踏む）。旧 heel_z の目では動かなかったので
#       露見していなかった。ビルド側と同じく continue する。
#   (3) sweep_limits.py の beak_y 軸が pf.beak_y（解析式の稜線）を測って
#       いて、稜線が胴の外へ出ても線形に伸び続けていた。部品の上面は
#       min(稜線, 外半径) なので、そこから先は形が動かない。
#
# §4〜§5 の一回きりのスクリプトは末尾に添付（リポジトリ直下に置いて
# PYTHONPATH=. python <file> で再現できる）。

"""B-4b プローブ: 3 本の壁が「壊れる点」よりどれだけ内側かを測る。

掃引器の 3 段は 3 本とも素通りする（`body_waist` ±0.36 / `shoulder_fullness`
±4.0 / `beak_height` +48.8〜53.1 まで ok）。壁を決めているのは形が壊れることでは
なく**主張が壊れる**ことなので、その距離を数字で残す。
"""
from saxsketch.core.registry import get_registry
from saxsketch.instruments.saxophone.components import profiles as pf

reg = get_registry()


def build_n(comp, p):
    try:
        part = comp.build(p)
        return len(part.solids()), part.volume
    except Exception as exc:                                     # noqa: BLE001
        return "RAISED %s" % type(exc).__name__, 0.0


for size in ("soprano", "alto", "tenor"):
    comp = reg.get("saxophone").make_component("mouthpiece", size)
    specs = {s.key: s for s in comp.param_specs()}
    base = {s.key: s.default for s in comp.param_specs()}
    bd = base["body_dia"]
    print("== %s  body_dia=%.1f" % (size, bd), flush=True)

    # --- 1) 頂点／えぐれが実際に壊れる点と、壁の位置
    for key, lim in (("body_waist", pf.body_waist_limits()),
                     ("shoulder_fullness", pf.shoulder_fullness_limits())):
        sp = specs[key]
        print("   %s: 帯 %+.3f〜%+.3f  壁 %+.3f〜%+.3f  解いた限界 %+.6f/%+.6f"
              % (key, sp.min, sp.max, sp.hard_min, sp.hard_max, lim[0], lim[1]),
              flush=True)
        for v in (lim[0] - 0.02, sp.hard_min, 0.0, sp.hard_max, lim[1] + 0.02):
            p = dict(base, **{key: v})
            az, ar = pf.outer_apex(p)
            dz, dr = pf.shoulder_dip(p)
            n, vol = build_n(comp, p)
            print("      %+8.4f  頂点 z=%8.3f (踵 %8.3f, ずれ %.3f)  "
                  "肩の最小 r=%.4f (シャンク %.4f)  ソリッド %s  体積 %.0f"
                  % (v, az, pf.heel_z(p), abs(az - pf.heel_z(p)), dr,
                     pf.shank_radius(p), n, vol), flush=True)

    # --- 2) beak_height: 壁の外で何が起きるか（定義が壊れる／OCCT が割れる）
    sp = specs["beak_height"]
    zb = pf._BEAK_BITE * base["overall_length"]
    print("   beak_height: 帯 %.1f〜%.1f  壁 %.1f〜%.1f  "
          "前下がり点 %.4f  胴の全高(bite) %.4f (%.4f x body_dia)"
          % (sp.min, sp.max, sp.hard_min, sp.hard_max,
             pf.beak_height_flat(base), pf.beak_height_at_body_top(base),
             pf.beak_height_at_body_top(base) / bd), flush=True)
    for v in (sp.hard_min - 0.5, sp.hard_min, base["beak_height"], sp.hard_max,
              sp.hard_max + 0.5, 2.0 * bd, -10.0, -19.0, -21.0):
        p = dict(base, beak_height=v)
        zs = [z for z in pf.beak_cut_stations(p) if 0.0 <= z <= pf.heel_z(p)]
        live = sum(1 for z in zs
                   if pf.beak_y(p, z) <= pf.outer_radius(p, z) + 1e-9)
        n, vol = build_n(comp, p)
        print("      %+8.2f (%.3f x bd)  ビーク角 %+6.2f deg  "
              "切除が効く断面 %2d/%2d  bite で稜線 %.3f vs 胴 %.3f  "
              "ソリッド %s  体積 %.0f"
              % (v, v / bd, __import__("math").degrees(pf.beak_angle(p)),
                 live, len(zs), pf.beak_y(p, zb), pf.outer_radius(p, zb),
                 n, vol), flush=True)

# -*- coding: utf-8 -*-
"""B-4b プローブ 2: 4 つの目を並べて出し、`beak_height` が割れる点を挟む。

プローブ 1 は `outer_apex` / `shoulder_dip` しか見ていなかったので、
肩比 +1.85〜+1.88 で**何も鳴らなかった**（膨らむが最大径は超えずシャンク径も
割らない）。単調性そのものを見る `body_dip` / `shoulder_bulge` を足して測り直す。

`beak_height` の下側は掃引器が打ち切り（−21mm）でしかビルドしておらず、
そこを「割れる点」と読むと嘘になる ——実際はもっと手前。0.5mm 刻みで挟む。
"""
import math

from saxsketch.core.registry import get_registry
from saxsketch.instruments.saxophone.components import profiles as pf

reg = get_registry()


def solids(comp, p):
    try:
        part = comp.build(p)
        return len(part.solids())
    except Exception as exc:                                     # noqa: BLE001
        return "RAISED %s" % type(exc).__name__


def eyes(p):
    az, _ = pf.outer_apex(p)
    _, dr = pf.shoulder_dip(p)
    _, drop = pf.body_dip(p)
    _, bulge = pf.shoulder_bulge(p)
    return (abs(az - pf.heel_z(p)), pf.shank_radius(p) - dr, drop, bulge)


for size in ("soprano", "alto", "tenor"):
    comp = reg.get("saxophone").make_component("mouthpiece", size)
    specs = {s.key: s for s in comp.param_specs()}
    base = {s.key: s.default for s in comp.param_specs()}
    print("== %s" % size, flush=True)
    print("   目: 頂点のずれ / えぐれ量 / 胴の凹み / 肩の膨らみ  [mm]",
          flush=True)
    for key, lim in (("body_waist", pf.body_waist_limits()),
                     ("shoulder_fullness", pf.shoulder_fullness_limits())):
        sp = specs[key]
        for v in (lim[0] - 0.02, sp.hard_min, 0.0, sp.hard_max, lim[1] + 0.02):
            a, b, c, d = eyes(dict(base, **{key: v}))
            rung = ["頂点", "えぐれ", "凹み", "膨らみ"]
            fired = [n for n, x in zip(rung, (a, b, c, d)) if x > 1e-9]
            print("      %-18s %+8.4f  %8.4f %8.4f %8.4f %8.4f   鳴った目: %s"
                  % (key, v, a, b, c, d, "／".join(fired) or "**どれも無言**"),
                  flush=True)

    # --- beak_height が割れる点を 0.5mm 刻みで挟む（壁は 4.5〜6.0）
    sp = specs["beak_height"]
    h = sp.hard_min
    last_ok = None
    while h > -12.0:
        n = solids(comp, dict(base, beak_height=h))
        if n != 1:
            print("   beak_height: 壁 %.1f から下って **%.1fmm でソリッド %s**"
                  "（直前 %.1f は 1 個）。ビーク角 %+.2f deg"
                  % (sp.hard_min, h, n, last_ok,
                     math.degrees(pf.beak_angle(dict(base, beak_height=h)))),
                  flush=True)
            break
        last_ok = h
        h = round(h - 0.5, 6)
    else:
        print("   beak_height: -12mm まで 1 個のまま", flush=True)
