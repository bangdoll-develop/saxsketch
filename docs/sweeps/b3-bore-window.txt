# Phase B-3 ボア 2 本の生ログ（2026-09-06、3 サイズ・OCCT ビルドあり）
# 読み方は docs/param-limits.md の「B-3 ボア ＋ ウィンドウ」。
# 掃引器: PYTHONPATH=. python cad/windycity/sweep_limits.py <key> [--steps N] [--with ...]
#
# このスライスで掃引器に足したもの: --steps（帯の外へ歩く段数）。
#   届く距離は「段数 × 刻み」なので、刻み 0.05mm の bore_dia は既定 40 段だと
#   帯の端から 2.0mm しか出られず、3 サイズとも「打ち切り」で終わる。
#   §1 がその打ち切り、§2 が --steps 200（10mm）で歩き直したもの。
#
# 掃引器だけでは決められなかったものを別に焼いてある:
#   §4 シャンクの肉厚 …… 段 3 は「ソリッド 1 個か」しか見ないので、
#      シャンクが丸ごと消えても ok と言う（体積だけが 61% 落ちる）。
#   §5 bore_length の両端 …… 上は掃引器の目（MIN_STATIONS 20 枚）が
#      モデルの成立条件（cavity_layout_error）より 6.5mm 手前で止まる。
#      下は段 1〜2 が −21mm まで素通りし、ビルドが Null shape で落ちる。
#   §6 入れた壁そのものをビルドしたもの（--no-build の結果は hard_* に入れない）。
#
# §4〜§6 の一回きりのスクリプトは末尾に添付（リポジトリ直下に置いて
# PYTHONPATH=. python <file> で再現できる）。

# -*- coding: utf-8 -*-
"""bore_dia の上側。**シャンクの肉厚**が実際にどこで壊れるかを 1 点ずつ測る。

掃引器の段 3 は「ソリッド 1 個 ＋ 体積 > 100mm3」しか見ないので、
**シャンクが消えても 1 個のまま**なら ok と言う（alto 26.2 / tenor 28.0 が
それ）。ここでは肉厚を並べて印字し、どこで割れるか・どこで印刷できなく
なるかを分けて見る。
"""
import sys, time
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parent))
sys.stdout.reconfigure(encoding="utf-8")

from saxsketch.core.registry import get_registry
from saxsketch.instruments.saxophone.components import profiles as pf

reg = get_registry()
for size in ("soprano", "alto", "tenor"):
    comp = reg.get("saxophone").make_component("mouthpiece", size)
    base = {s.key: s.default for s in comp.param_specs()}
    sh = base["shank_dia"]; mw = base["min_wall"]
    spec = next(s for s in comp.param_specs() if s.key == "bore_dia")
    print(f"== {size} == shank_dia φ{sh} / min_wall {mw} / 帯 {spec.min}〜{spec.max}"
          f" / 既定 φ{base['bore_dia']}  肉厚が min_wall になるのは φ{sh - 2*mw:.2f}",
          flush=True)
    for d in (spec.max, sh - 2*mw, sh - 2*mw + 0.5, sh - 2.0, sh - 1.0, sh - 0.4,
              sh, sh + 0.4, sh + 1.0, sh + 2.0):
        p = dict(base, bore_dia=round(d, 3))
        wall = (sh - d) / 2.0
        err = comp.params_error(p)
        t0 = time.time()
        try:
            part = comp.build(p)
            got = f"ソリッド {len(part.solids())} 個 / 体積 {part.volume:9.1f}mm3"
        except Exception as exc:
            got = f"RAISED {type(exc).__name__}: {exc}"[:60]
        print(f"   φ{d:6.2f}  肉厚 {wall:+5.2f}mm  params_error={'None' if err is None else err[:30]}"
              f"  {got}  ({time.time()-t0:.1f}s)", flush=True)

----- b3_probe_len.py -----
# -*- coding: utf-8 -*-
"""bore_length の両端。**掃引器の目とモデルの成立条件は別の場所にある。**

上: 掃引器は `MIN_STATIONS`（20 枚）で止まるが、モデル自身が弾くのは
    `cavity_layout_error`（窓 2.0mm）。どちらが先かを数える。
下: cheap（段 1〜2）は負まで通るのにビルドが Null shape で落ちる。
    どこから落ちるかを 1mm ずつ測る。
"""
import sys, time
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parent))
sys.stdout.reconfigure(encoding="utf-8")

from saxsketch.core.registry import get_registry
from saxsketch.instruments.saxophone.components import profiles as pf

reg = get_registry()
for size in ("soprano", "alto", "tenor"):
    comp = reg.get("saxophone").make_component("mouthpiece", size)
    base = {s.key: s.default for s in comp.param_specs()}
    spec = next(s for s in comp.param_specs() if s.key == "bore_length")
    L = base["overall_length"]
    print(f"== {size} == 全長 {L} / 帯 {spec.min}〜{spec.max} / 既定 {base['bore_length']}",
          flush=True)

    # --- 上: モデルの成立条件 vs 掃引器の目。3 通りの相手で。
    for tag, extra in (("既定", {}), ("window_length=44", {"window_length": 44.0}),
                       (f"overall_length={next(s for s in comp.param_specs() if s.key=='overall_length').min:g}",
                        {"overall_length": next(s for s in comp.param_specs()
                                                if s.key == "overall_length").min})):
        b = dict(base, **extra)
        err_at = st_at = None
        for i in range(0, 1200):
            v = spec.min + i * 0.1
            p = dict(b, bore_length=v)
            if err_at is None and comp.params_error(p) is not None:
                err_at = v
            if st_at is None:
                try:
                    if len(pf.cavity_stations(p)) < 20:
                        st_at = v
                except Exception:
                    st_at = v
            if err_at is not None and st_at is not None:
                break
        print(f"   上／{tag}: モデルが弾くのは {err_at}mm（zb={b['overall_length']-err_at:.1f}）"
              f" / 掃引器の目（20 枚）は {st_at}mm", flush=True)

    # --- 下: ビルドが落ちる位置。
    for v in (2.0, 1.0, 0.0, -1.0, -1.4, -1.5, -2.0):
        p = dict(base, bore_length=v)
        err = comp.params_error(p)
        t0 = time.time()
        try:
            part = comp.build(p)
            got = f"ソリッド {len(part.solids())} 個 / 体積 {part.volume:8.1f}mm3"
        except Exception as exc:
            got = f"RAISED {type(exc).__name__}: {exc}"[:50]
        print(f"   下／bore_length {v:+5.1f}  zb={L - v:6.1f}  "
              f"params_error={'None' if err is None else err[:28]}  {got}  "
              f"({time.time()-t0:.1f}s)", flush=True)

"""入れようとしている壁の値そのものをビルドする。

`--no-build` の結果を `hard_*` に入れてはいけない（掃引器の docstring）。
掃引の段 3 は「掃引器が止まった点」の周りしか焼いていないので、

    base = {s.key: s.default for s in comp.param_specs()}
    L, sh = base["overall_length"], base["shank_dia"]
    hhi_len = round(L - 6.5, 2)
    hhi_dia = round(sh - 0.05, 2)
    print(f"== {size} == 全長 {L} / シャンク φ{sh}", flush=True)
    for key, vals in (("bore_length", (hhi_len - 1.0, hhi_len, hhi_len + 1.0)),

            err = comp.params_error(p)
            n = None

                n = f"{type(exc).__name__}"

                try:
                    part = comp.build(p)
